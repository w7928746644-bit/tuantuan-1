💡 首次开启多端云同步说明（喂养 + 生长曲线 + 自定义按钮 + 每周评分）

当前应用已内置您的 Supabase 密钥，本地随时可用。如需爸爸妈妈多部手机实时同步喂养、身高体重、自定义按钮及每周评分历史，请在 Supabase 控制台的 SQL Editor 中运行下方建表脚本： 

create table if not exists public.feeding_records (
  id text primary key,
  family_code text not null default 'tuantuan-family-01',
  record_date text not null,
  record_time text not null,
  timestamp bigint not null,
  category text not null,
  title text not null,
  detail text,
  left_min int default 0,
  right_min int default 0,
  formula_ml int default 0,
  recorder text default '妈妈',
  created_at timestamptz default now()
);
create table if not exists public.growth_logs (
  id text primary key,
  family_code text not null default 'tuantuan-family-01',
  measure_date text not null,
  timestamp bigint not null,
  weight_kg numeric(5,2),
  height_cm numeric(5,1),
  head_cm numeric(4,1),
  note text,
  recorder text default '妈妈',
  created_at timestamptz default now()
);
create table if not exists public.family_button_configs (
  family_code text primary key,
  custom_buttons jsonb not null default '[]'::jsonb,
  updated_at bigint not null default 0,
  updated_by text default '妈妈',
  updated_iso timestamptz default now()
);
create table if not exists public.weekly_scores (
  id text primary key,
  family_code text not null default 'tuantuan-family-01',
  week_key text not null,
  week_label text not null,
  total_score int not null,
  grade_label text not null,
  dimensions jsonb not null default '[]'::jsonb,
  summary_text text,
  updated_at bigint not null default 0,
  updated_by text default '妈妈',
  created_at timestamptz default now()
);
alter table public.feeding_records enable row level security;
alter table public.growth_logs enable row level security;
alter table public.family_button_configs enable row level security;
alter table public.weekly_scores enable row level security;
create policy "Allow public access feeding" on public.feeding_records for all using (true) with check (true);
create policy "Allow public access growth" on public.growth_logs for all using (true) with check (true);
create policy "Allow public access button configs" on public.family_button_configs for all using (true) with check (true);
create policy "Allow public access weekly scores" on public.weekly_scores for all using (true) with check (true);
alter publication supabase_realtime add table public.feeding_records;
alter publication supabase_realtime add table public.growth_logs;
alter publication supabase_realtime add table public.family_button_configs;
alter publication supabase_realtime add table public.weekly_scores;
